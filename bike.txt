    <script>
        const API_URL = 'https://tcgbusfs.blob.core.windows.net/dotapp/youbike/v2/youbike_immediate.json';
        let map, userMarker, markersLayer;
        let userCoords = [25.04085, 121.5116]; // 預設：台北車站

        // 初始化地圖
        function initMap() {
            map = L.map('map').setView(userCoords, 15);
            L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
                attribution: '© OpenStreetMap contributors'
            }).addTo(map);
            markersLayer = L.layerGroup().addTo(map);
        }

        // 計算距離 (Haversine formula)
        function getDistance(lat1, lon1, lat2, lon2) {
            if (lat1 === undefined || lon1 === undefined || lat2 === undefined || lon2 === undefined) return Infinity;
            const R = 6371; // km
            const dLat = (lat2 - lat1) * Math.PI / 180;
            const dLon = (lon2 - lon1) * Math.PI / 180;
            const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
                      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon/2) * Math.sin(dLon/2);
            const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
            return R * c * 1000; // 返回公尺
        }

        // 取得定位
        function getUserLocation() {
            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(
                    (pos) => {
                        userCoords = [pos.coords.latitude, pos.coords.longitude];
                        map.setView(userCoords, 16);
                        updateUserMarker();
                        fetchData();
                    },
                    (err) => {
                        console.warn("定位失敗:", err.message);
                        updateUserMarker();
                        fetchData();
                    }
                );
            } else {
                updateUserMarker();
                fetchData();
            }
        }

        function updateUserMarker() {
            if (userMarker) map.removeLayer(userMarker);
            const icon = L.divIcon({ className: 'user-location-icon', iconSize: [16, 16] });
            userMarker = L.marker(userCoords, { icon }).addTo(map).bindPopup("<b>您的目前位置</b>").openPopup();
        }

        // 取得數據並渲染
        async function fetchData() {
            const loading = document.getElementById('loadingStatus');
            const icon = document.getElementById('refreshIcon');
            loading.classList.remove('hidden');
            icon.classList.add('animate-spin');

            try {
                const res = await fetch(API_URL);
                const data = await res.json();
                
                // 計算每個站點距離使用者的距離，增加對 lat/lng 命名不同的容錯
                const stationsWithDist = data.map(s => {
                    const sLat = Number(s.latitude || s.lat);
                    const sLng = Number(s.longitude || s.lng);
                    return {
                        ...s,
                        computedLat: sLat,
                        computedLng: sLng,
                        dist: getDistance(userCoords[0], userCoords[1], sLat, sLng)
                    };
                }).filter(s => !isNaN(s.computedLat) && !isNaN(s.computedLng)); // 過濾掉座標無效的站點

                // 排序並取前 50 個顯示在地圖上
                stationsWithDist.sort((a, b) => a.dist - b.dist);
                
                renderMarkers(stationsWithDist.slice(0, 50));
                renderList(stationsWithDist.slice(0, 6));
                document.getElementById('dataTime').innerText = `同步時間: ${new Date().toLocaleTimeString()}`;
            } catch (err) {
                console.error("資料讀取錯誤:", err);
            } finally {
                loading.classList.add('hidden');
                icon.classList.remove('animate-spin');
            }
        }

        function renderMarkers(stations) {
            markersLayer.clearLayers();
            stations.forEach(s => {
                const available = Number(s.available_rent_bikes || s.sbi || 0);
                const returnSlots = Number(s.available_return_bikes || s.available_return_slots || s.bemp || 0);
                const color = available > 5 ? '#10b981' : (available > 0 ? '#f59e0b' : '#ef4444');
                
                // 確保座標存在才繪製
                if (s.computedLat && s.computedLng) {
                    const marker = L.circleMarker([s.computedLat, s.computedLng], {
                        radius: 8,
                        fillColor: color,
                        color: "#fff",
                        weight: 2,
                        opacity: 1,
                        fillOpacity: 0.8
                    });

                    const popupContent = `
                        <div class="custom-popup p-2">
                            <h4 class="font-black text-slate-800 text-lg">${s.sna.replace('YouBike2.0_', '')}</h4>
                            <p class="text-slate-400 text-xs mb-3">${s.ar}</p>
                            <div class="grid grid-cols-2 gap-2">
                                <div class="bg-green-50 p-2 rounded-xl text-center">
                                    <span class="block text-[8px] font-bold text-green-600">可借車輛</span>
                                    <span class="text-xl font-black text-green-700">${available}</span>
                                </div>
                                <div class="bg-blue-50 p-2 rounded-xl text-center">
                                    <span class="block text-[8px] font-bold text-blue-600">可還空位</span>
                                    <span class="text-xl font-black text-blue-700">${returnSlots}</span>
                                </div>
                            </div>
                        </div>
                    `;
                    marker.bindPopup(popupContent);
                    markersLayer.addLayer(marker);
                }
            });
        }

        function renderList(stations) {
            const container = document.getElementById('nearList');
            container.innerHTML = '';
            stations.forEach(s => {
                const available = Number(s.available_rent_bikes || s.sbi || 0);
                const returnSlots = Number(s.available_return_bikes || s.available_return_slots || s.bemp || 0);
                const quantity = Number(s.Quantity || s.tot || 0);
                const updateTime = s.mday ? `${s.mday.slice(8,10)}:${s.mday.slice(10,12)}` : "--:--";

                const card = document.createElement('div');
                card.className = "bg-white p-6 rounded-[2rem] shadow-sm border border-slate-100 transition-all hover:shadow-md cursor-pointer";
                card.onclick = () => map.setView([s.computedLat, s.computedLng], 18);
                
                card.innerHTML = `
                    <div class="flex justify-between items-start mb-4">
                        <span class="px-3 py-1 bg-slate-100 text-slate-500 text-[10px] font-black rounded-lg uppercase">距離 ${s.dist === Infinity ? '未知' : Math.round(s.dist) + 'm'}</span>
                        <span class="text-[10px] text-slate-300 font-bold">${updateTime}</span>
                    </div>
                    <h3 class="font-black text-slate-800 text-lg mb-1">${s.sna.replace('YouBike2.0_', '')}</h3>
                    <p class="text-slate-400 text-xs mb-6 truncate">${s.ar}</p>
                    
                    <div class="grid grid-cols-2 gap-3 mb-4">
                        <div class="bg-emerald-50 rounded-2xl p-3 text-center">
                            <p class="text-[9px] font-bold text-emerald-600 mb-1">可借車輛</p>
                            <p class="text-2xl font-black text-emerald-700">${available}</p>
                        </div>
                        <div class="bg-blue-50 rounded-2xl p-3 text-center">
                            <p class="text-[9px] font-bold text-blue-600 mb-1">可還空位</p>
                            <p class="text-2xl font-black text-blue-700">${returnSlots}</p>
                        </div>
                    </div>
                    <div class="bg-amber-50 rounded-xl p-2 text-center text-amber-700 text-xs font-bold">
                        總格數 (Qty): ${quantity}
                    </div>
                `;
                container.appendChild(card);
            });
        }

        // 初始化
        window.onload = () => {
            initMap();
            getUserLocation();
        };

        document.getElementById('refreshBtn').addEventListener('click', fetchData);
        document.getElementById('locateBtn').addEventListener('click', getUserLocation);
    </script>

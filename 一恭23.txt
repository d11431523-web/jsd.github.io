    <script>
        const canvas = document.getElementById('main-canvas');
        const ctx = canvas.getContext('2d', { willReadFrequently: true });
        const cropOverlay = document.getElementById('crop-overlay');
        const modeInfo = document.getElementById('mode-info');
        
        let state = {
            baseImg: null,
            currentImg: null,
            history: [], // 歷史紀錄棧
            mode: 'idle',
            isDrawing: false,
            startX: 0, startY: 0,
            settings: { brightness: 100, contrast: 100, temp: 0 },
            brushSize: 30
        };

        function getCoord(e) {
            const rect = canvas.getBoundingClientRect();
            const scaleX = canvas.width / rect.width;
            const scaleY = canvas.height / rect.height;
            return {
                realX: (e.clientX - rect.left) * scaleX,
                realY: (e.clientY - rect.top) * scaleY,
                cssX: e.clientX - rect.left,
                cssY: e.clientY - rect.top
            };
        }

        // 核心渲染：渲染傳入的影像物件
        function render(imgObj) {
            if (!imgObj) return;
            canvas.width = imgObj.width;
            canvas.height = imgObj.height;
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.drawImage(imgObj, 0, 0);
        }

        // 提交當前狀態到歷史紀錄
        function commit(label) {
            const dataUrl = canvas.toDataURL();
            const img = new Image();
            img.onload = () => {
                state.currentImg = img;
                state.history.push({ 
                    label, 
                    src: dataUrl, 
                    settings: { ...state.settings } 
                });
                updateHistoryUI();
            };
            img.src = dataUrl;
        }

        function updateHistoryUI() {
            const log = document.getElementById('history-log');
            log.innerHTML = '';
            state.history.slice().reverse().forEach(h => {
                const item = document.createElement('div');
                item.className = 'history-item';
                item.innerText = h.label;
                log.appendChild(item);
            });
        }

        document.getElementById('upload-input').onchange = (e) => {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = (ev) => {
                const img = new Image();
                img.onload = () => {
                    state.baseImg = img;
                    state.history = [];
                    // 初始渲染
                    render(img);
                    // 初始 commit 作為第一筆紀錄
                    commit("載入原始影像");
                    document.getElementById('main-controls').classList.remove('app-disabled');
                    document.getElementById('empty-state').style.display = 'none';
                    canvas.style.display = 'block';
                };
                img.src = ev.target.result;
            };
            reader.readAsDataURL(file);
        };

        function setMode(newMode, infoText) {
            document.querySelectorAll('.btn-action').forEach(b => b.classList.remove('active'));
            if (state.mode === newMode) {
                state.mode = 'idle';
                modeInfo.style.display = 'none';
                cropOverlay.style.display = 'none';
            } else {
                state.mode = newMode;
                if (infoText) {
                    modeInfo.innerText = infoText;
                    modeInfo.style.display = 'block';
                }
                const activeBtn = document.querySelector(`[data-mode="${newMode}"]`) || document.getElementById('tool-watermark');
                if (activeBtn) activeBtn.classList.add('active');
            }
        }

        document.getElementById('tool-chromakey').onclick = () => setMode('chromakey', '去背模式：點擊想要移除的顏色');
        document.getElementById('tool-crop').onclick = () => setMode('crop', '裁切模式：拖曳範圍並釋放');
        document.getElementById('tool-watermark').onclick = () => setMode('watermark', '浮水印模式：點擊位置放置文字');

        document.getElementById('brush-size').oninput = (e) => state.brushSize = parseInt(e.target.value);

        let compareTimer = null;
        let isLongPress = false;

        canvas.onmousedown = (e) => {
            if (state.mode === 'idle') {
                compareTimer = setTimeout(() => {
                    isLongPress = true;
                    render(state.baseImg);
                    modeInfo.innerText = "對比原圖中...";
                    modeInfo.style.display = 'block';
                }, 250);
                return;
            }

            const pos = getCoord(e);
            state.isDrawing = true;

            if (state.mode === 'chromakey') {
                removeColorAt(pos.realX, pos.realY);
                commit("顏色去背");
                setMode('idle');
            } else if (state.mode === 'watermark') {
                addWatermarkAt(pos.realX, pos.realY);
                commit("放置浮水印");
                setMode('idle');
            } else if (state.mode === 'crop') {
                state.startX = pos.cssX;
                state.startY = pos.cssY;
                cropOverlay.style.display = 'block';
                cropOverlay.style.left = pos.cssX + 'px';
                cropOverlay.style.top = pos.cssY + 'px';
                cropOverlay.style.width = '0px';
                cropOverlay.style.height = '0px';
            }
        };

        window.onmousemove = (e) => {
            if (!state.isDrawing) return;
            const pos = getCoord(e);

            if (state.mode === 'crop') {
                const w = pos.cssX - state.startX;
                const h = pos.cssY - state.startY;
                cropOverlay.style.width = Math.abs(w) + 'px';
                cropOverlay.style.height = Math.abs(h) + 'px';
                cropOverlay.style.left = (w > 0 ? state.startX : pos.cssX) + 'px';
                cropOverlay.style.top = (h > 0 ? state.startY : pos.cssY) + 'px';
            }
        };

        window.onmouseup = (e) => {
            clearTimeout(compareTimer);
            if (isLongPress) {
                render(state.currentImg);
                modeInfo.style.display = 'none';
                isLongPress = false;
            }

            if (!state.isDrawing) return;
            state.isDrawing = false;

            if (state.mode === 'crop') {
                executeCrop();
            }
        };

        function removeColorAt(x, y) {
            const idata = ctx.getImageData(0, 0, canvas.width, canvas.height);
            const d = idata.data;
            const targetIdx = (Math.floor(y) * canvas.width + Math.floor(x)) * 4;
            const tr = d[targetIdx], tg = d[targetIdx+1], tb = d[targetIdx+2];
            for(let i = 0; i < d.length; i += 4) {
                const diff = Math.sqrt(Math.pow(d[i]-tr,2)+Math.pow(d[i+1]-tg,2)+Math.pow(d[i+2]-tb,2));
                if(diff < 45) d[i+3] = 0;
            }
            ctx.putImageData(idata, 0, 0);
        }

        function addWatermarkAt(x, y) {
            const text = document.getElementById('watermark-text').value || "WORKSHOP";
            const color = document.getElementById('watermark-color').value;
            ctx.font = `bold ${state.brushSize}px Inter, sans-serif`;
            ctx.fillStyle = color;
            ctx.textAlign = 'center';
            ctx.fillText(text, x, y);
        }

        function executeCrop() {
            const crect = canvas.getBoundingClientRect();
            const orect = cropOverlay.getBoundingClientRect();
            const scaleX = canvas.width / crect.width;
            const scaleY = canvas.height / crect.height;
            const x = (orect.left - crect.left) * scaleX;
            const y = (orect.top - crect.top) * scaleY;
            const w = orect.width * scaleX;
            const h = orect.height * scaleY;
            if (w < 10 || h < 10) return;
            const temp = document.createElement('canvas');
            temp.width = w; temp.height = h;
            temp.getContext('2d').drawImage(canvas, x, y, w, h, 0, 0, w, h);
            const ni = new Image();
            ni.onload = () => {
                render(ni);
                commit("影像裁切");
                setMode('idle');
                modeInfo.style.display = 'none';
                cropOverlay.style.display = 'none';
            };
            ni.src = temp.toDataURL();
        }

        // 處理基礎調整
        function applyBaseFilters() {
            // 從 currentImg (上一步提交的影像) 重新渲染
            render(state.currentImg);
            const idata = ctx.getImageData(0, 0, canvas.width, canvas.height);
            const d = idata.data;
            const b = state.settings.brightness / 100;
            const c = state.settings.contrast / 100;
            const t = state.settings.temp;
            for(let i = 0; i < d.length; i += 4) {
                for(let j = 0; j < 3; j++) d[i+j] = ((d[i+j]/255-0.5)*c+0.5)*255*b;
                d[i] += t; d[i+2] -= t;
            }
            ctx.putImageData(idata, 0, 0);
        }

        ['bright', 'contrast', 'temp'].forEach(k => {
            const el = document.getElementById(`adj-${k}`);
            el.oninput = () => {
                const val = parseInt(el.value);
                state.settings[k === 'bright' ? 'brightness' : k === 'contrast' ? 'contrast' : 'temp'] = val;
                document.getElementById(`v-${k}`).innerText = val;
                applyBaseFilters();
            };
            el.onchange = () => commit(`調整基礎參數`);
        });

        // 修正後的復原邏輯
        document.getElementById('btn-undo').onclick = () => {
            // 必須保證歷史紀錄至少有 1 筆
            if (state.history.length > 1) {
                // 移除當前最新的那一筆
                state.history.pop();
                // 取得移除後的最末位作為當前影像
                const lastNode = state.history[state.history.length - 1];
                const img = new Image();
                img.onload = () => {
                    state.currentImg = img;
                    state.settings = { ...lastNode.settings };
                    // 同步滑桿 UI
                    document.getElementById('adj-bright').value = state.settings.brightness;
                    document.getElementById('adj-contrast').value = state.settings.contrast;
                    document.getElementById('adj-temp').value = state.settings.temp;
                    document.getElementById('v-bright').innerText = state.settings.brightness;
                    document.getElementById('v-contrast').innerText = state.settings.contrast;
                    document.getElementById('v-temp').innerText = state.settings.temp;
                    render(img);
                };
                img.src = lastNode.src;
            } else {
                // 如果只剩一筆，就不再 pop，保持當前狀態
                console.log("已到達最初狀態");
            }
            setMode('idle');
            updateHistoryUI();
        };

        document.getElementById('btn-reset').onclick = () => {
            render(state.baseImg);
            state.history = [];
            state.settings = { brightness: 100, contrast: 100, temp: 0 };
            // 更新滑桿數值
            document.getElementById('adj-bright').value = 100;
            document.getElementById('adj-contrast').value = 100;
            document.getElementById('adj-temp').value = 0;
            document.getElementById('v-bright').innerText = 100;
            document.getElementById('v-contrast').innerText = 100;
            document.getElementById('v-temp').innerText = 0;
            commit("手動全部重置");
            setMode('idle');
        };

        document.getElementById('btn-download').onclick = () => {
            const link = document.createElement('a');
            link.download = `EXPORT_${Date.now()}.png`;
            link.href = canvas.toDataURL();
            link.click();
        };

        document.getElementById('drop-zone').onclick = () => document.getElementById('upload-input').click();
    </script>
